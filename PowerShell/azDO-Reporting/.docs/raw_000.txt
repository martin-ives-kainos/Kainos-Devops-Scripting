Complete Project Inventory Report
Import-Module ImportExcel

$OrgUrl = "https://dev.azure.com/MyOrganisation"

# Get all projects
$Projects = az devops project list `
    --org $OrgUrl `
    --output json | ConvertFrom-Json

$Report = foreach ($Project in $Projects.value)
{
    Write-Host "Processing $($Project.name)..."

    # Repositories
    $Repos = az repos list `
        --project $Project.name `
        --org $OrgUrl `
        --output json 2>$null | ConvertFrom-Json

    # Pipelines
    $Pipelines = az pipelines list `
        --project $Project.name `
        --org $OrgUrl `
        --output json 2>$null | ConvertFrom-Json

    # Teams
    $Teams = az devops team list `
        --project $Project.name `
        --org $OrgUrl `
        --output json 2>$null | ConvertFrom-Json

    [PSCustomObject]@{
        ProjectName      = $Project.name
        Id               = $Project.id
        Description      = $Project.description
        Visibility       = $Project.visibility
        State            = $Project.state
        Url              = $Project.url
        Revision         = $Project.revision
        RepositoryCount  = @($Repos).Count
        PipelineCount    = @($Pipelines).Count
        TeamCount        = @($Teams).Count
    }
}

$ExcelFile = ".\AzureDevOps-ProjectReport.xlsx"

$Report |
    Export-Excel `
        -Path $ExcelFile `
        -WorksheetName Projects `
        -TableName Projects `
        -FreezeTopRow `
        -BoldTopRow `
        -AutoSize `
        -AutoFilter

Write-Host "Report created: $ExcelFile"

Enhanced Report with Process, Service Connections and Variable Groups

Some details require REST API calls.

$PAT = $env:AZDO_PAT

$Base64PAT = :ToBase64String(
    [Text.Encoding]::ASCII.GetBytes(":$PAT")
)

$Headers = @{
    Authorization = "Basic $Base64PAT"
}

Service Connections Count
$EndpointsUri =
"$OrgUrl/$($Project.name)/_apis/serviceendpoint/endpoints?api-version=7.1"

$EndpointCount = (
    Invoke-RestMethod `
        -Uri $EndpointsUri `
        -Headers $Headers
).value.Count

Variable Groups Count
$VarGroupUri =
"$OrgUrl/$($Project.name)/_apis/distributedtask/variablegroups?api-version=7.1-preview.2"

$VarGroupCount = (
    Invoke-RestMethod `
        -Uri $VarGroupUri `
        -Headers $Headers
).value.Count

Process Template
$ProjectDetails = az devops project show `
    --project $Project.name `
    --org $OrgUrl `
    --output json | ConvertFrom-Json

$ProcessName = $ProjectDetails.capabilities.processTemplate.templateName

Add formatting:

$excel = Open-ExcelPackage $ExcelFile

foreach($ws in $excel.Workbook.Worksheets)
{
    $ws.Cells.AutoFitColumns()
    $ws.View.FreezePanes(2,1)

    Add-ConditionalFormatting `
        -Worksheet $ws `
        -Address "A2:Z5000" `
        -RuleType Expression `
        -ConditionValue 'MOD(ROW(),2)=0'
}

Close-ExcelPackage $excel
